\begin{frame}[t]
  \frametitle{Funktionsweise der Anwendung}

  \begin{itemize}
    \item[\faKey] \textbf{Zugang:} Sysadmins geben Zugriff (evtl.\ in R\"ucksprache)
    \medskip
    \item[\faUser] \textbf{Datenfelder allgemein:}\\
      {\small Name, Vorname, Titel, Abschluss, Gender, RZ-Kennung,
       E-Mail, Telefon, Geb\"aude/Raum, Homepage, Kommentar (Homepage / intern),
       \glqq auf Homepage sichtbar\grqq}
    \medskip
    \item[\faLock] \textbf{Nur Dekanat:}\\
      {\small Einstiegs- und Ausstiegsdatum, Kommentar zur Stelle,
       l\"angere Abwesenheiten}
    \medskip
    \item[\faTags] \textbf{\glqq Codes\grqq} werden Personen zugeordnet,\\
      {\small z.B.\ \glqq Abteilung Logik\grqq{} oder \glqq deutschsprachig\grqq{}
       \quad \textcolor{gray}{(Kategorien: Statusgruppe, Abteilung, Sprache, \dots)}}
  \end{itemize}
\end{frame}

\begin{frame}[t]
  \frametitle{Such- und Export-Funktion}

  \begin{itemize}
    \setlength{\itemsep}{6pt}
    \item[\faSearch] Suche nach Personen mit \textbf{allen} angegebenen Codes
    \item[\faCalendar] Stichtag w\"ahlbar: wer ist zu diesem Datum am Institut?
    \item[\faTable] Auswahl der ausgegebenen Spalten (z.B.\ Name, Mail, Abteilung, \dots)
    \item[\faFileExcel] Export als Excel-Datei
    \item[\textcolor{middle_red}{\faExclamationTriangle}] \textbf{Der Verlauf wird nicht gespeichert.}\\
      {\small Gespeichert wird nur der aktuelle Stand (plus letzter Bearbeitungsstempel).}
  \end{itemize}
\end{frame}
